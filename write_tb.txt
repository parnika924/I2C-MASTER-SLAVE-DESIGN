module i2c_m_tb;

reg clk, rst_n, start;
reg [6:0] add;
reg [7:0] data;

wire scl;
wire sda;
wire busy;

// pull-up resistor model for SDA
pullup(sda);

i2c_m M1 (
    .clk(clk),
    .rst_n(rst_n),
    .start(start),
    .add(add),
    .data(data),
    .scl(scl),
    .sda(sda),
    .busy(busy)
);

initial
begin
    clk = 0;
    forever #5 clk = ~clk;
end

initial
begin
    rst_n = 0;
    start = 0;
    add   = 7'h42;
    data  = 8'hA5;

    #20 rst_n = 1;

    #30 start = 1;   // triggers transaction
    #10 start = 0;

    #500 $finish;
end

initial
begin
    $dumpfile("i2c_m.vcd");
    $dumpvars(0, i2c_m_tb);
end

endmodule
