// I2C Master to Slave (Write)

module i2c_m (
    input wire clk,
    input wire rst_n,
    input wire start,
    input [6:0] add,
    input [7:0] data,
    output reg scl,
    inout sda,
    output reg busy
);

reg [3:0] state;
reg [3:0] bit_cnt;     // bit control
reg sda_out, sda_oe;

assign sda = sda_oe ? sda_out : 1'bz;   // Open drain SDA

localparam
    IDLE  = 0,
    START = 1,
    ADDR  = 2,
    ACK1  = 3,
    DATA  = 4,
    ACK2  = 5,
    STOP  = 6;

always @(posedge clk or negedge rst_n)
begin
    // asynchronous active low reset
    if (!rst_n)
    begin
        state   <= IDLE;
        scl     <= 1'b1;
        sda_out <= 1'b1;
        sda_oe  <= 1'b1;
        busy    <= 1'b0;
        bit_cnt <= 0;
    end
    else
    begin
        case(state)

        IDLE:
        begin
            scl     <= 1'b1;
            sda_out <= 1'b1;
            busy    <= 1'b0;

            if (start)
            begin
                busy  <= 1'b1;
                state <= START;
            end
        end

        START:
        begin
            // start condition
            sda_out <= 1'b0;
            state   <= ADDR;
            bit_cnt <= 7;
        end

        ADDR:
        begin
            scl     <= 1'b0;
            sda_out <= add[bit_cnt];

            scl <= 1'b1;

            if (bit_cnt == 0)
            begin
                state   <= ACK1;
                sda_out <= 1'b0; // write bit = 0
            end
            else
                bit_cnt <= bit_cnt - 1;
        end

        ACK1:
        begin
            scl    <= 1'b0;
            sda_oe <= 1'b0; // release SDA for ACK

            scl <= 1'b1;

            state   <= DATA;
            bit_cnt <= 7;
            sda_oe  <= 1'b1;
        end

        DATA:
        begin
            scl     <= 1'b0;
            sda_out <= data[bit_cnt];

            scl <= 1'b1;

            if (bit_cnt == 0)
                state <= ACK2;
            else
                bit_cnt <= bit_cnt - 1;
        end

        ACK2:
        begin
            scl    <= 1'b0;
            sda_oe <= 1'b0; // release SDA for ACK

            scl <= 1'b1;

            state  <= STOP;
            sda_oe <= 1'b1;
        end

        STOP:
        begin
            scl     <= 1'b1;
            sda_out <= 1'b0;
            sda_oe  <= 1'b1; // stop condition

            state <= IDLE;
            busy  <= 1'b0;
        end

        endcase
    end
end

endmodule
